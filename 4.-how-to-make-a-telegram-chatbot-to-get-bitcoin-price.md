* TGbot\_1.py

```
import json
import requests
import time
import urllib

TOKEN = "1617421892:AAFG51PHXYeq34rSKJ-nUvdDEcjI-FMkQYo"
URL = "https://api.telegram.org/bot{}/".format(TOKEN)


def get_url(url):
    response = requests.get(url)    
    content = response.content.decode("utf8")    
    return content
   
     
def get_json_from_url(url):
    content = get_url(url)    
    js = json.loads(content)    
    return js
   
     
def get_updates(offset=None):
    url = URL + "getUpdates?timeout=100"    
    if offset:        
        url += "&offset={}".format(offset)    
    js = get_json_from_url(url)    
    return js
  
      
def get_last_update_id(updates):
    update_ids = []    
    for update in updates["result"]:
        update_ids.append(int(update["update_id"]))    
    return max(update_ids)
    
    
def echo_all(updates):
    for update in updates["result"]:
        text = update["message"]["text"]
        msg = text    
        try:
            chat = update["message"]["chat"]["id"]
            send_message(msg, chat)        
        except Exception as e:     
            print(e)


def get_last_chat_id_and_text(updates):
    num_updates = len(updates["result"])
        last_update = num_updates - 1    
        text = updates["result"][last_update]["message"]["text"]    
        chat_id = updates["result"][last_update]["message"]["chat"]["id"]    
        return (text, chat_id)
        
def send_message(text, chat_id):    
    text = urllib.parse.quote_plus(text)    
    url = URL + "sendMessage?text={}&chat_id={}".format(text, chat_id)    
    get_url(url)


def main():
    last_update_id = None    
    while True:        
        updates = get_updates(last_update_id)        
        if len(updates["result"]) > 0:            
            last_update_id = get_last_update_id(updates) + 1            
            echo_all(updates)        
        time.sleep(0.5)
        
if __name__ == '__main__':
    main()
```

## How to use ChatBot to get Bitcoin price

Our requirements: If we sent `/reply` to Chatbox, it will get the latest bitcoin price for us, and reply the message we sent back to us in other circumstance.

We can simply edit the reply logics in our bot, so our bot can almost do anything for us! Simply edit the function `echo_all` to the following:

```
def echo_all(updates):
    for update in updates["result"]:    
        text = update["message"]["text"]        
        if text == '/price':     
               price = ftx.fetch_order_book('BTC/USD', 10)['bids'][0][0]      
               msg = 'BTC price %s' % price        
        else:            
               msg = text        
        try:            
               chat = update["message"]["chat"]["id"]       
               send_message(msg, chat)    
        except Exception as e:            
               print(e)
```

and also add two lines at the top of the program:

```
import ccxt
ftx = ccxt.ftx()
```
